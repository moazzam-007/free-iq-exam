    function loadBestScore() {
      var saved = localStorage.getItem('fiq_game_nback_best');
      if (saved) {
        hudBestScore.textContent = saved;
      }
    }
    loadBestScore();

    function updateHud() {
      hudNLevel.textContent = currentN + '-Back';
      hudTrialCount.textContent = trialIndex + ' / ' + TOTAL_TRIALS;
      hudScore.textContent = score;
      if (overlayNText) {
        overlayNText.textContent = currentN + ' steps back';
      }
    }

    function showFeedback(text, isPositive) {
      if (!feedbackToast) return;
      feedbackToast.textContent = text;
      feedbackToast.className = 'h-6 mt-4 text-xs font-mono font-semibold transition-opacity ' +
        (isPositive ? 'text-emerald-600 dark:text-emerald-400' : 'text-rose-600 dark:text-rose-400');
      feedbackToast.style.opacity = '1';
      setTimeout(function() {
        feedbackToast.style.opacity = '0';
      }, 700);
    }

    function generateTrialSequence() {
      trials = [];
      for (var i = 0; i < TOTAL_TRIALS; i++) {
        var pos;
        var letter;

        // Position decision
        if (i >= currentN && Math.random() < 0.35) {
          pos = trials[i - currentN].pos;
        } else {
          var availPos = [0, 1, 2, 3, 4, 5, 6, 7, 8];
          if (i >= currentN) {
            availPos = availPos.filter(function(p) { return p !== trials[i - currentN].pos; });
          }
          pos = availPos[Math.floor(Math.random() * availPos.length)];
        }

        // Letter decision
        if (i >= currentN && Math.random() < 0.35) {
          letter = trials[i - currentN].letter;
        } else {
          var availLetters = LETTERS.slice();
          if (i >= currentN) {
            availLetters = availLetters.filter(function(l) { return l !== trials[i - currentN].letter; });
          }
          letter = availLetters[Math.floor(Math.random() * availLetters.length)];
        }

        trials.push({ pos: pos, letter: letter });
      }
    }

    function highlightCell(idx, active) {
      var cell = document.getElementById('cell-' + idx);
      if (!cell) return;
      if (active) {
        cell.className = 'rounded-xl border-2 border-blue-500 bg-blue-500 shadow-lg shadow-blue-500/50 flex items-center justify-center scale-105 transition-all duration-150';
        cell.innerHTML = '<div class=\"w-3.5 h-3.5 rounded-full bg-white animate-pulse\"></div>';
      } else {
        cell.className = 'rounded-xl border border-zinc-200 dark:border-zinc-700/60 bg-white dark:bg-zinc-900 transition-all duration-200 flex items-center justify-center';
        cell.innerHTML = '<div class=\"w-2.5 h-2.5 rounded-full bg-zinc-300 dark:bg-zinc-700\"></div>';
      }
    }

    function recordPositionInput() {
      if (!isRunning || pressedPos) return;
      pressedPos = true;
      btnPosMatch.classList.add('ring-4', 'ring-blue-400', 'bg-blue-600', 'text-white');

      if (trialIndex > currentN) {
        var current = trials[trialIndex - 1];
        var past = trials[trialIndex - 1 - currentN];
        var isMatch = current.pos === past.pos;
        if (isMatch) {
          score += 10;
          correctPosMatches++;
          showFeedback('+10 Position Match!', true);
        } else {
          score = Math.max(0, score - 5);
          falsePosMatches++;
          showFeedback('-5 False Alarm!', false);
        }
        hudScore.textContent = score;
      }
    }

    function recordLetterInput() {
      if (!isRunning || pressedLetter) return;
      pressedLetter = true;
      btnLetterMatch.classList.add('ring-4', 'ring-indigo-400', 'bg-indigo-600', 'text-white');

      if (trialIndex > currentN) {
        var current = trials[trialIndex - 1];
        var past = trials[trialIndex - 1 - currentN];
        var isMatch = current.letter === past.letter;
        if (isMatch) {
          score += 10;
          correctLetterMatches++;
          showFeedback('+10 Letter Match!', true);
        } else {
          score = Math.max(0, score - 5);
          falseLetterMatches++;
          showFeedback('-5 False Alarm!', false);
        }
        hudScore.textContent = score;
      }
    }

    function evaluateMisses(evaluatedIndex) {
      if (evaluatedIndex <= currentN) return;
      var cur = trials[evaluatedIndex - 1];
      var past = trials[evaluatedIndex - 1 - currentN];

      var posMatched = cur.pos === past.pos;
      var letterMatched = cur.letter === past.letter;

      // Opportunities count
      totalDecisions += 2;

      // Position evaluation
      if (posMatched && pressedPos) {
        correctDecisions++;
      } else if (!posMatched && !pressedPos) {
        correctDecisions++;
      }

      // Letter evaluation
      if (letterMatched && pressedLetter) {
        correctDecisions++;
      } else if (!letterMatched && !pressedLetter) {
        correctDecisions++;
      }
    }

    function resetInputButtons() {
      pressedPos = false;
      pressedLetter = false;
      btnPosMatch.className = 'py-4 px-4 rounded-2xl bg-blue-50 dark:bg-blue-950/40 border-2 border-blue-500 text-blue-700 dark:text-blue-300 font-heading font-bold text-sm sm:text-base transition-all duration-100 hover:bg-blue-100 dark:hover:bg-blue-900/50 active:scale-95 flex flex-col items-center justify-center shadow-sm';
      btnLetterMatch.className = 'py-4 px-4 rounded-2xl bg-indigo-50 dark:bg-indigo-950/40 border-2 border-indigo-500 text-indigo-700 dark:text-indigo-300 font-heading font-bold text-sm sm:text-base transition-all duration-100 hover:bg-indigo-100 dark:hover:bg-indigo-900/50 active:scale-95 flex flex-col items-center justify-center shadow-sm';
    }

    function runNextTrial() {
      if (trialIndex > 0) {
        evaluateMisses(trialIndex);
      }

      if (trialIndex >= TOTAL_TRIALS) {
        finishRound();
        return;
      }

      resetInputButtons();
      var t = trials[trialIndex];
      trialIndex++;
      updateHud();

      // Show stimulus
      highlightCell(t.pos, true);
      stimulusLetter.textContent = t.letter;
      stimulusLetter.style.opacity = '1';

      // Hide stimulus after STIMULUS_DURATION
      clearStimulusTimer = setTimeout(function() {
        highlightCell(t.pos, false);
        stimulusLetter.textContent = '-';
        stimulusLetter.style.opacity = '0.3';
      }, STIMULUS_DURATION);

      trialTimer = setTimeout(runNextTrial, TRIAL_INTERVAL);
    }

    function startRound() {
      if (isRunning) return;
      isRunning = true;
      trialIndex = 0;
      score = 0;
      correctPosMatches = 0;
      correctLetterMatches = 0;
      falsePosMatches = 0;
      falseLetterMatches = 0;
      totalDecisions = 0;
      correctDecisions = 0;

      generateTrialSequence();
      updateHud();

      startOverlay.classList.add('hidden');
      modalEndRound.classList.add('hidden');
      modalEndRound.classList.remove('flex');

      runNextTrial();
    }

    function finishRound() {
      isRunning = false;
      clearTimeout(trialTimer);
      clearTimeout(clearStimulusTimer);
      resetInputButtons();

      var accuracy = totalDecisions > 0 ? Math.round((correctDecisions / totalDecisions) * 100) : 0;

      var prevN = currentN;
      var statusMsg = '';
      if (accuracy >= 80) {
        currentN++;
        statusMsg = 'Promoted! Advanced to ' + currentN + '-Back based on ' + accuracy + '% accuracy.';
      } else if (accuracy < 60) {
        currentN = Math.max(1, currentN - 1);
        statusMsg = 'Regressed to ' + currentN + '-Back to consolidate working memory baseline.';
      } else {
        statusMsg = 'Maintained ' + currentN + '-Back. Target 80%+ accuracy to level up.';
      }

      var savedBest = parseInt(localStorage.getItem('fiq_game_nback_best') || '0', 10);
      var isNewBest = score > savedBest;
      if (isNewBest) {
        localStorage.setItem('fiq_game_nback_best', score.toString());
        loadBestScore();
      }

      modalScore.textContent = score;
      modalAccuracy.textContent = accuracy + '%';
      modalNextN.textContent = currentN + '-Back';
      modalLevelStatus.textContent = statusMsg;

      if (isNewBest && score > 0) {
        modalBestAlert.classList.remove('hidden');
      } else {
        modalBestAlert.classList.add('hidden');
      }

      modalEndRound.classList.remove('hidden');
      modalEndRound.classList.add('flex');
    }

    // Button event listeners
    btnStartGame.addEventListener('click', startRound);
    btnPosMatch.addEventListener('click', recordPositionInput);
    btnLetterMatch.addEventListener('click', recordLetterInput);
    btnModalNextRound.addEventListener('click', function() {
      modalEndRound.classList.add('hidden');
      modalEndRound.classList.remove('flex');
      startRound();
    });

    // Keyboard shortcuts
    window.addEventListener('keydown', function(e) {
      if (e.repeat) return;
      var key = e.key.toLowerCase();
      if (key === ' ' && !isRunning && modalEndRound.classList.contains('hidden')) {
        e.preventDefault();
        startRound();
      } else if (key === 'a') {
        recordPositionInput();
      } else if (key === 'l') {
        recordLetterInput();
      }
    });
  })();
</script>

    /**
     * Programmatically generate isometric SVG cube
     * faces: [topFaceIdx, leftFaceIdx, rightFaceIdx]
     * modifiedFaceIndex: optional index of face with altered inset color
     */
    function renderCubeSvg(faces, modifiedFaceIndex, alteredColorHex) {
      // Shading styles for isometric 3D illusion
      // Top face: lighter, Left: base, Right: darker
      var baseShades = {
        top: '#cbd5e1', // Light slate
        left: '#94a3b8', // Medium slate
        right: '#64748b' // Dark slate
      };

      var topColor = (modifiedFaceIndex === 0 && alteredColorHex) ? alteredColorHex : PALETTES[faces[0]].hex;
      var leftColor = (modifiedFaceIndex === 1 && alteredColorHex) ? alteredColorHex : PALETTES[faces[1]].hex;
      var rightColor = (modifiedFaceIndex === 2 && alteredColorHex) ? alteredColorHex : PALETTES[faces[2]].hex;

      var svg = [
        '<svg viewBox=\"0 0 220 220\" class=\"w-full h-full drop-shadow-md\" xmlns=\"http://www.w3.org/2000/svg\">',
        '  <g stroke=\"#334155\" stroke-width=\"2\" stroke-linejoin=\"round\">',
        '    <!-- Top Face -->',
        '    <polygon points=\"110,45 170,80 110,115 50,80\" fill=\"' + baseShades.top + '\" />',
        '    <!-- Top Inset Rhombus Marker -->',
        '    <polygon points=\"110,62 142,80 110,98 78,80\" fill=\"' + topColor + '\" stroke=\"rgba(0,0,0,0.15)\" stroke-width=\"1.5\" />',
        '    <!-- Left Face -->',
        '    <polygon points=\"50,80 110,115 110,185 50,150\" fill=\"' + baseShades.left + '\" />',
        '    <!-- Left Inset Rhombus Marker -->',
        '    <polygon points=\"65,103 95,120 95,155 65,138\" fill=\"' + leftColor + '\" stroke=\"rgba(0,0,0,0.15)\" stroke-width=\"1.5\" />',
        '    <!-- Right Face -->',
        '    <polygon points=\"110,115 170,80 170,150 110,185\" fill=\"' + baseShades.right + '\" />',
        '    <!-- Right Inset Rhombus Marker -->',
        '    <polygon points=\"125,120 155,103 155,138 125,155\" fill=\"' + rightColor + '\" stroke=\"rgba(0,0,0,0.15)\" stroke-width=\"1.5\" />',
        '  </g>',
        '</svg>'
      ].join('');

      return svg;
    }

    function createTrial() {
      // Pick base rotation for Cube A
      var rotAIdx = Math.floor(Math.random() * VALID_ROTATIONS.length);
      var facesA = VALID_ROTATIONS[rotAIdx];

      var isSame = Math.random() < 0.5;

      // Pick a different rotation for Cube B
      var rotBIdx = (rotAIdx + 1 + Math.floor(Math.random() * (VALID_ROTATIONS.length - 1))) % VALID_ROTATIONS.length;
      var facesB = VALID_ROTATIONS[rotBIdx];

      var modifiedFace = -1;
      var alteredColor = null;

      if (!isSame) {
        // Pick one face of Cube B to modify
        modifiedFace = Math.floor(Math.random() * 3);
        // Pick distinct color not currently on that face
        var currentFaceId = facesB[modifiedFace];
        var distinctColors = ['#ef4444', '#8b5cf6', '#14b8a6', '#f97316'].filter(function(c) {
          return c !== PALETTES[currentFaceId].hex;
        });
        alteredColor = distinctColors[Math.floor(Math.random() * distinctColors.length)];
      }

      return {
        facesA: facesA,
        facesB: facesB,
        isSame: isSame,
        modifiedFace: modifiedFace,
        alteredColor: alteredColor
      };
    }

    function startTimerLoop() {
      trialStartTime = performance.now();
      var circumference = 88;

      function step() {
        if (!isRunning) return;
        var elapsed = performance.now() - trialStartTime;
        var remaining = Math.max(0, TRIAL_TIME_LIMIT_MS - elapsed);
        var fraction = remaining / TRIAL_TIME_LIMIT_MS;

        timerText.textContent = (remaining / 1000).toFixed(1) + 's';
        timerRingArc.style.strokeDashoffset = (circumference * (1 - fraction)).toString();

        if (remaining <= 2000) {
          timerText.classList.add('text-rose-600', 'dark:text-rose-400');
          timerRingArc.classList.add('text-rose-600', 'dark:text-rose-400');
        } else {
          timerText.classList.remove('text-rose-600', 'dark:text-rose-400');
          timerRingArc.classList.remove('text-rose-600', 'dark:text-rose-400');
        }

        if (remaining > 0) {
          animFrame = requestAnimationFrame(step);
        } else {
          handleTimeout();
        }
      }

      animFrame = requestAnimationFrame(step);
    }

    function handleTimeout() {
      cancelAnimationFrame(animFrame);
      showFeedback('Timeout (0 pts)', false);
      responseTimes.push(TRIAL_TIME_LIMIT_MS);
      nextTrial();
    }

    function handleAnswer(userSame) {
      if (!isRunning) return;
      cancelAnimationFrame(animFrame);
      var responseTime = performance.now() - trialStartTime;
      responseTimes.push(responseTime);

      var isCorrect = userSame === currentTrialData.isSame;
      if (isCorrect) {
        correctCount++;
        var pts = 0;
        if (responseTime <= 3000) {
          pts = 20;
        } else if (responseTime <= 6000) {
          pts = 10;
        } else {
          pts = 5;
        }
        score += pts;
        showFeedback('Correct! +' + pts + ' pts (' + (responseTime / 1000).toFixed(2) + 's)', true);
      } else {
        showFeedback('Incorrect (0 pts)', false);
      }

      nextTrial();
    }

    function nextTrial() {
      trialIndex++;
      updateHud();

      if (trialIndex >= TOTAL_TRIALS) {
        finishGame();
        return;
      }

      currentTrialData = createTrial();
      cubeAContainer.innerHTML = renderCubeSvg(currentTrialData.facesA, -1, null);
      cubeBContainer.innerHTML = renderCubeSvg(currentTrialData.facesB, currentTrialData.modifiedFace, currentTrialData.alteredColor);

      startTimerLoop();
    }

    function startGame() {
      isRunning = true;
      trialIndex = 0;
      score = 0;
      correctCount = 0;
      responseTimes = [];

      startOverlay.classList.add('hidden');
      modalEndRound.classList.add('hidden');
      modalEndRound.classList.remove('flex');

      updateHud();
      nextTrial();
    }

    function finishGame() {
      isRunning = false;
      cancelAnimationFrame(animFrame);

      var accuracy = Math.round((correctCount / TOTAL_TRIALS) * 100);
      var sum = 0;
      for (var i = 0; i < responseTimes.length; i++) sum += responseTimes[i];
      var avgSpeed = (sum / responseTimes.length / 1000).toFixed(2) + 's';

      var savedBest = parseInt(localStorage.getItem(STORAGE_KEY) || '0', 10);
      var isNewBest = score > savedBest;
      if (isNewBest) {
        localStorage.setItem(STORAGE_KEY, score.toString());
        loadBestScore();
      }

      modalScore.textContent = score;
      modalAcc.textContent = accuracy + '%';
      modalSpeed.textContent = avgSpeed;

      if (isNewBest && score > 0) {
        modalBestAlert.classList.remove('hidden');
      } else {
        modalBestAlert.classList.add('hidden');
      }

      modalEndRound.classList.remove('hidden');
      modalEndRound.classList.add('flex');
    }

    // Event listeners
    btnStartGame.addEventListener('click', startGame);
    btnSame.addEventListener('click', function() { handleAnswer(true); });
    btnDiff.addEventListener('click', function() { handleAnswer(false); });
    btnModalRestart.addEventListener('click', function() {
      modalEndRound.classList.add('hidden');
      modalEndRound.classList.remove('flex');
      startGame();
    });

    window.addEventListener('keydown', function(e) {
      if (e.repeat) return;
      var key = e.key.toLowerCase();
      if (key === ' ' && !isRunning && modalEndRound.classList.contains('hidden')) {
        e.preventDefault();
        startGame();
      } else if (key === 's') {
        handleAnswer(true);
      } else if (key === 'd') {
        handleAnswer(false);
      }
    });
  })();
</script>

      var tier = getTier();
      hudLevel.textContent = 'Tier ' + tier;
      hudStreak.textContent = streak;
      hudScore.textContent = score;
    }

    function showFeedback(text, isPositive) {
      if (!feedbackToast) return;
      feedbackToast.textContent = text;
      feedbackToast.className = 'h-6 mt-4 text-xs font-mono font-semibold transition-opacity ' +
        (isPositive ? 'text-emerald-600 dark:text-emerald-400' : 'text-rose-600 dark:text-rose-400');
      feedbackToast.style.opacity = '1';
      setTimeout(function() {
        feedbackToast.style.opacity = '0';
      }, 600);
    }

    function generateProblem() {
      var tier = getTier();
      var expr = '';
      var ans = 0;

      if (tier === 1) {
        // Level 1: Single-digit addition/subtraction
        var op = Math.random() < 0.5 ? '+' : '-';
        if (op === '+') {
          var a = Math.floor(Math.random() * 8) + 2; // 2..9
          var b = Math.floor(Math.random() * 8) + 2;
          expr = a + ' + ' + b;
          ans = a + b;
        } else {
          var b = Math.floor(Math.random() * 8) + 2;
          var res = Math.floor(Math.random() * 8) + 1;
          var a = b + res;
          expr = a + ' - ' + b;
          ans = res;
        }
      } else if (tier === 2) {
        // Level 2: Two-digit +/- single-digit, or single-digit product
        var roll = Math.random();
        if (roll < 0.4) {
          var a = Math.floor(Math.random() * 70) + 15;
          var b = Math.floor(Math.random() * 8) + 2;
          expr = a + ' + ' + b;
          ans = a + b;
        } else if (roll < 0.7) {
          var a = Math.floor(Math.random() * 70) + 20;
          var b = Math.floor(Math.random() * 8) + 2;
          expr = a + ' - ' + b;
          ans = a - b;
        } else {
          var a = Math.floor(Math.random() * 7) + 3; // 3..9
          var b = Math.floor(Math.random() * 8) + 2;
          expr = a + ' × ' + b;
          ans = a * b;
        }
      } else if (tier === 3) {
        // Level 3: Two-digit * single-digit, 2-digit +/- 2-digit, mixed
        var roll = Math.random();
        if (roll < 0.4) {
          var a = Math.floor(Math.random() * 15) + 11; // 11..25
          var b = Math.floor(Math.random() * 6) + 3; // 3..8
          expr = a + ' × ' + b;
          ans = a * b;
        } else if (roll < 0.7) {
          var a = Math.floor(Math.random() * 6) + 3;
          var b = Math.floor(Math.random() * 5) + 2;
          var c = Math.floor(Math.random() * 5) + 2;
          expr = a + ' × (' + b + ' + ' + c + ')';
          ans = a * (b + c);
        } else {
          var a = Math.floor(Math.random() * 50) + 30;
          var b = Math.floor(Math.random() * 25) + 15;
          expr = a + ' - ' + b;
          ans = a - b;
        }
      } else {
        // Level 4: Mental division, squares, combined
        var roll = Math.random();
        if (roll < 0.35) {
          var b = Math.floor(Math.random() * 7) + 3; // 3..9
          var res = Math.floor(Math.random() * 12) + 6; // 6..17
          var a = b * res;
          var c = Math.floor(Math.random() * 12) + 3;
          expr = a + ' ÷ ' + b + ' + ' + c;
          ans = res + c;
        } else if (roll < 0.7) {
          var a = Math.floor(Math.random() * 8) + 11; // 11..18
          expr = a + '²';
          ans = a * a;
        } else {
          var b = Math.floor(Math.random() * 7) + 3;
          var res = Math.floor(Math.random() * 10) + 4;
          var a = b * res;
          expr = a + ' ÷ ' + b;
          ans = res;
        }
      }

      // Generate 3 unique plausible distractors
      var options = [ans];
      var offsets = [-1, 1, -2, 2, -3, 3, -10, 10, -5, 5, -4, 4];
      offsets.sort(function() { return 0.5 - Math.random(); });

      for (var i = 0; i < offsets.length; i++) {
        var cand = ans + offsets[i];
        if (cand > 0 && options.indexOf(cand) === -1) {
          options.push(cand);
        }
        if (options.length === 4) break;
      }

      // If still less than 4, fill
      var extra = 1;
      while (options.length < 4) {
        var cand = ans + extra;
        if (options.indexOf(cand) === -1) options.push(cand);
        extra += 2;
      }

      // Shuffle options
      options.sort(function() { return 0.5 - Math.random(); });

      return {
        expr: expr,
        ans: ans,
        options: options
      };
    }

    function renderProblem() {
      currentProblem = generateProblem();
      equationText.textContent = currentProblem.expr + ' = ?';

      for (var i = 0; i < 4; i++) {
        optTexts[i].textContent = currentProblem.options[i];
        optButtons[i].className = 'py-4 px-5 rounded-2xl bg-zinc-50 dark:bg-zinc-800 border-2 border-zinc-200 dark:border-zinc-700 hover:border-emerald-500 dark:hover:border-emerald-500 text-zinc-900 dark:text-zinc-100 font-heading font-bold text-xl sm:text-2xl transition-all duration-100 hover:scale-102 active:scale-95 flex items-center justify-between shadow-sm';
      }
    }

    function handleChoice(optIdx) {
      if (!isRunning || !currentProblem) return;
      var chosenVal = currentProblem.options[optIdx];
      var btn = optButtons[optIdx];

      if (chosenVal === currentProblem.ans) {
        score += 10;
        correctCount++;
        streak++;
        if (streak > longestStreak) longestStreak = streak;

        // Check for 5-streak bonus
        if (streak % 5 === 0) {
          score += 5;
          streakBonusBadge.textContent = streak + 'x Streak (+5 Bonus)';
          streakBonusBadge.classList.remove('hidden');
          setTimeout(function() {
            streakBonusBadge.classList.add('hidden');
          }, 1200);
          showFeedback('Streak Bonus! +15 pts', true);
        } else {
          showFeedback('Correct! +10 pts', true);
        }

        btn.classList.add('border-emerald-500', 'bg-emerald-50', 'dark:bg-emerald-950/40');
      } else {
        score = Math.max(0, score - 3);
        wrongCount++;
        streak = 0;
        showFeedback('Wrong! -3 pts (Answer: ' + currentProblem.ans + ')', false);
        btn.classList.add('border-rose-500', 'bg-rose-50', 'dark:bg-rose-950/40');
      }

      updateHud();
      setTimeout(function() {
        if (isRunning) renderProblem();
      }, 120);
    }

    function timerStep() {
      if (!isRunning) return;
      var elapsed = performance.now() - startTime;
      var remaining = Math.max(0, TOTAL_DURATION_MS - elapsed);
      var fraction = remaining / TOTAL_DURATION_MS;

      sprintTimerText.textContent = (remaining / 1000).toFixed(1) + 's';
      sprintProgressBar.style.width = (fraction * 100).toFixed(1) + '%';

      if (remaining <= 10000) {
        sprintTimerText.classList.add('text-rose-600', 'dark:text-rose-400', 'animate-pulse');
        sprintProgressBar.classList.remove('bg-emerald-500');
        sprintProgressBar.classList.add('bg-rose-500');
      } else {
        sprintTimerText.classList.remove('text-rose-600', 'dark:text-rose-400', 'animate-pulse');
        sprintProgressBar.classList.add('bg-emerald-500');
        sprintProgressBar.classList.remove('bg-rose-500');
      }

      if (remaining > 0) {
        timerAnimFrame = requestAnimationFrame(timerStep);
      } else {
        finishSprint();
      }
    }

    function startSprint() {
      isRunning = true;
      score = 0;
      streak = 0;
      longestStreak = 0;
      correctCount = 0;
      wrongCount = 0;
      startTime = performance.now();

      startOverlay.classList.add('hidden');
      modalEndRound.classList.add('hidden');
      modalEndRound.classList.remove('flex');

      updateHud();
      renderProblem();
      timerAnimFrame = requestAnimationFrame(timerStep);
    }

    function finishSprint() {
      isRunning = false;
      cancelAnimationFrame(timerAnimFrame);

      var totalAttempted = correctCount + wrongCount;
      var accuracy = totalAttempted > 0 ? Math.round((correctCount / totalAttempted) * 100) : 0;

      var savedBest = parseInt(localStorage.getItem(STORAGE_KEY) || '0', 10);
      var isNewBest = score > savedBest;
      if (isNewBest) {
        localStorage.setItem(STORAGE_KEY, score.toString());
        loadBestScore();
      }

      modalScore.textContent = score;
      modalAcc.textContent = accuracy + '%';
      modalAttempted.textContent = totalAttempted;
      modalStreak.textContent = longestStreak;

      if (isNewBest && score > 0) {
        modalBestAlert.classList.remove('hidden');
      } else {
        modalBestAlert.classList.add('hidden');
      }

      modalEndRound.classList.remove('hidden');
      modalEndRound.classList.add('flex');
    }

    // Bind option buttons
    optButtons.forEach(function(btn, idx) {
      btn.addEventListener('click', function() { handleChoice(idx); });
    });

    btnStartGame.addEventListener('click', startSprint);
    btnModalRestart.addEventListener('click', function() {
      modalEndRound.classList.add('hidden');
      modalEndRound.classList.remove('flex');
      startSprint();
    });

    // Keyboard controls
    window.addEventListener('keydown', function(e) {
      if (e.repeat) return;
      var key = e.key;
      if (key === ' ' && !isRunning && modalEndRound.classList.contains('hidden')) {
        e.preventDefault();
        startSprint();
      } else if (key === '1') {
        handleChoice(0);
      } else if (key === '2') {
        handleChoice(1);
      } else if (key === '3') {
        handleChoice(2);
      } else if (key === '4') {
        handleChoice(3);
      }
    });
  })();
</script>

    function updateHud() {
      hudRound.textContent = roundIndex + ' / ' + TOTAL_ROUNDS;
      hudScore.textContent = score;
      if (responseTimes.length > 0) {
        var sum = 0;
        for (var i = 0; i < responseTimes.length; i++) sum += responseTimes[i];
        var avg = (sum / responseTimes.length / 1000).toFixed(2);
        hudAvgTime.textContent = avg + 's';
      } else {
        hudAvgTime.textContent = '0.0s';

    function showFeedback(text, isPositive) {
      if (!feedbackToast) return;
      feedbackToast.textContent = text;
      feedbackToast.className = 'h-6 mt-2 text-xs font-mono font-semibold transition-opacity ' +
        (isPositive ? 'text-emerald-600 dark:text-emerald-400' : 'text-rose-600 dark:text-rose-400');
      feedbackToast.style.opacity = '1';

    function getShapeSvg(shape, colorHex) {
      var inner = '';
      if (shape === 'circle') {
        inner = '<circle cx=\"50\" cy=\"50\" r=\"36\" fill=\"' + colorHex + '\" />';
      } else if (shape === 'square') {
        inner = '<rect x=\"16\" y=\"16\" width=\"68\" height=\"68\" rx=\"10\" fill=\"' + colorHex + '\" />';
      } else if (shape === 'triangle') {
        inner = '<polygon points=\"50,14 88,82 12,82\" fill=\"' + colorHex + '\" />';
      } else if (shape === 'diamond') {
        inner = '<polygon points=\"50,12 88,50 50,88 12,50\" fill=\"' + colorHex + '\" />';
      } else if (shape === 'pentagon') {
        inner = '<polygon points=\"50,14 86,40 72,84 28,84 14,40\" fill=\"' + colorHex + '\" />';
      } else if (shape === 'hexagon') {
        inner = '<polygon points=\"50,14 84,33 84,71 50,90 16,71 16,33\" fill=\"' + colorHex + '\" />';
      } else if (shape === 'star') {
        inner = '<polygon points=\"50,12 62,38 88,38 67,54 75,80 50,64 25,80 33,54 12,38 38,38\" fill=\"' + colorHex + '\" />';
      } else if (shape === 'cross') {
        inner = '<polygon points=\"36,14 64,14 64,36 86,36 86,64 64,64 64,86 36,86 36,64 14,64 14,36 36,36\" fill=\"' + colorHex + '\" />';
      } else if (shape === 'arrow-up') {
        inner = '<polygon points=\"50,12 86,48 64,48 64,86 36,86 36,48 14,48\" fill=\"' + colorHex + '\" />';
      } else if (shape === 'arrow-right') {
        inner = '<polygon points=\"88,50 52,14 52,36 14,36 14,64 52,64 52,86\" fill=\"' + colorHex + '\" />';
      } else if (shape === 'oval') {
        inner = '<ellipse cx=\"50\" cy=\"50\" rx=\"42\" ry=\"26\" fill=\"' + colorHex + '\" />';
      } else if (shape === 'ring') {
        inner = '<path fill-rule=\"evenodd\" d=\"M50 14a36 36 0 100 72 36 36 0 000-72zm0 16a20 20 0 110 40 20 20 0 010-40z\" fill=\"' + colorHex + '\" />';
      }
      return '<svg viewBox=\"0 0 100 100\" class=\"w-full h-full drop-shadow-sm\" xmlns=\"http://www.w3.org/2000/svg\">' + inner + '</svg>';
    }

    function createRound() {
      // Pick random target shape & color
      var targetShape = SHAPES[Math.floor(Math.random() * SHAPES.length)];
      var targetColor = COLORS[Math.floor(Math.random() * COLORS.length)];

      var otherColors = COLORS.filter(function(c) { return c.hex !== targetColor.hex; });
      otherColors.sort(function() { return 0.5 - Math.random(); });

      var otherShapes = SHAPES.filter(function(s) { return s !== targetShape; });
      otherShapes.sort(function() { return 0.5 - Math.random(); });

      // Generate the 6 candidates:
      // 1 exact match
      var c0 = { shape: targetShape, color: targetColor, isCorrect: true };
      // 2 same shape, wrong color
      var c1 = { shape: targetShape, color: otherColors[0], isCorrect: false };
      var c2 = { shape: targetShape, color: otherColors[1], isCorrect: false };
      // 2 wrong shape, same color
      var c3 = { shape: otherShapes[0], color: targetColor, isCorrect: false };
      var c4 = { shape: otherShapes[1], color: targetColor, isCorrect: false };
      // 1 completely different
      var c5 = { shape: otherShapes[2], color: otherColors[2], isCorrect: false };

      var options = [c0, c1, c2, c3, c4, c5];
      options.sort(function() { return 0.5 - Math.random(); });

      var correctOptionIndex = -1;
      for (var i = 0; i < options.length; i++) {
        if (options[i].isCorrect) {
          correctOptionIndex = i;
          break;
        }
      }

      return {
        targetShape: targetShape,
        targetColor: targetColor,
        options: options,
        correctOptionIndex: correctOptionIndex
      };
    }

    function renderRound() {
      currentRoundData = createRound();

      // Render target
      targetSymbolContainer.innerHTML = getShapeSvg(currentRoundData.targetShape, currentRoundData.targetColor.hex);

      // Render 6 options
      for (var i = 0; i < 6; i++) {
        var opt = currentRoundData.options[i];
        optContainers[i].innerHTML = getShapeSvg(opt.shape, opt.color.hex);
        optButtons[i].className = 'group relative flex flex-col items-center justify-center p-3 rounded-2xl bg-zinc-50 dark:bg-zinc-800/70 border-2 border-zinc-200 dark:border-zinc-700 hover:border-purple-500 dark:hover:border-purple-400 shadow-sm transition-all duration-150 hover:scale-105 active:scale-95 aspect-square';
      }

      startTrialTimer();
    }

    function startTrialTimer() {
      roundStartTime = performance.now();

      function step() {
        if (!isRunning) return;
        var elapsed = performance.now() - roundStartTime;
        var remaining = Math.max(0, ROUND_TIME_LIMIT_MS - elapsed);
        var fraction = remaining / ROUND_TIME_LIMIT_MS;

        trialTimerText.textContent = (remaining / 1000).toFixed(1) + 's';
        trialProgressBar.style.width = (fraction * 100).toFixed(1) + '%';

        if (remaining <= 1000) {
          trialTimerText.classList.add('text-rose-600', 'dark:text-rose-400');
          trialProgressBar.classList.remove('bg-purple-500');
          trialProgressBar.classList.add('bg-rose-500');
        } else {
          trialTimerText.classList.remove('text-rose-600', 'dark:text-rose-400');
          trialProgressBar.classList.add('bg-purple-500');
          trialProgressBar.classList.remove('bg-rose-500');
        }

        if (remaining > 0) {
          timerAnimFrame = requestAnimationFrame(step);
        } else {
          handleTimeout();
        }
      }

      timerAnimFrame = requestAnimationFrame(step);
    }

    function handleTimeout() {
      cancelAnimationFrame(timerAnimFrame);
      responseTimes.push(ROUND_TIME_LIMIT_MS);
      showFeedback('Timeout (0 pts)', false);
      nextRound();
    }

    function handleSelect(optIdx) {
      if (!isRunning || !currentRoundData) return;
      cancelAnimationFrame(timerAnimFrame);

      var elapsed = performance.now() - roundStartTime;
      responseTimes.push(elapsed);

      var btn = optButtons[optIdx];
      if (optIdx === currentRoundData.correctOptionIndex) {
        correctCount++;
        var pts = 0;
        if (elapsed < 1000) {
          pts = 25;
        } else if (elapsed < 2000) {
          pts = 15;
        } else {
          pts = 10;
        }
        score += pts;
        showFeedback('Exact Match! +' + pts + ' pts (' + (elapsed / 1000).toFixed(2) + 's)', true);
        btn.classList.add('border-emerald-500', 'bg-emerald-50', 'dark:bg-emerald-950/40');
      } else {
        score = Math.max(0, score - 5);
        showFeedback('Wrong Match! -5 pts', false);
        btn.classList.add('border-rose-500', 'bg-rose-50', 'dark:bg-rose-950/40');
      }

      updateHud();
      setTimeout(nextRound, 120);
    }

    function nextRound() {
      roundIndex++;
      updateHud();

      if (roundIndex >= TOTAL_ROUNDS) {
        finishBlitz();
        return;
      }

      renderRound();
    }

    function startBlitz() {
      isRunning = true;
      roundIndex = 0;
      score = 0;
      correctCount = 0;
      responseTimes = [];

      startOverlay.classList.add('hidden');
      modalEndRound.classList.add('hidden');
      modalEndRound.classList.remove('flex');

      updateHud();
      nextRound();
    }

    function finishBlitz() {
      isRunning = false;
      cancelAnimationFrame(timerAnimFrame);

      var accuracy = Math.round((correctCount / TOTAL_ROUNDS) * 100);
      var sum = 0;
      for (var i = 0; i < responseTimes.length; i++) sum += responseTimes[i];
      var avgSpeed = (sum / responseTimes.length / 1000).toFixed(2) + 's';

      var savedBest = parseInt(localStorage.getItem(STORAGE_KEY) || '0', 10);
      var isNewBest = score > savedBest;
      if (isNewBest) {
        localStorage.setItem(STORAGE_KEY, score.toString());
        loadBestScore();
      }

      modalScore.textContent = score;
      modalAcc.textContent = accuracy + '%';
      modalSpeed.textContent = avgSpeed;

      if (isNewBest && score > 0) {
        modalBestAlert.classList.remove('hidden');
      } else {
        modalBestAlert.classList.add('hidden');
      }

      modalEndRound.classList.remove('hidden');
      modalEndRound.classList.add('flex');
    }

    // Bind option buttons
    optButtons.forEach(function(btn, idx) {
      btn.addEventListener('click', function() { handleSelect(idx); });
    });

    btnStartGame.addEventListener('click', startBlitz);
    btnModalRestart.addEventListener('click', function() {
      modalEndRound.classList.add('hidden');
      modalEndRound.classList.remove('flex');
      startBlitz();
    });

    // Keyboard shortcuts 1..6
    window.addEventListener('keydown', function(e) {
      if (e.repeat) return;
      var key = e.key;
      if (key === ' ' && !isRunning && modalEndRound.classList.contains('hidden')) {
        e.preventDefault();
        startBlitz();
      } else if (key >= '1' && key <= '6') {
        var idx = parseInt(key, 10) - 1;
        handleSelect(idx);
      }
    });
  })();
</script>

    function finishDrill() {
      isRunning = false;
      cancelAnimationFrame(timerAnimFrame);

      var accuracy = Math.round((correctCount / TOTAL_ITEMS) * 100);
      var sum = 0;
